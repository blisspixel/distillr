class TrialLimitReached(BudgetExceededError):
    def __init__(self, message):
        super().__init__(0, 0)
        self.args = (message,)


def remaining():
    left = DEADLINE - time.monotonic()
    if left < 1 or len(ATTEMPTS) >= 24:
        raise TrialLimitReached('Authorized trial request/time limit reached')
    return left


def save():
    RECEIPT['elapsed_seconds'] = round(time.monotonic() - START, 3)
    atomic_write_json(ROOT / 'trial.json', RECEIPT)


original_connect = socket.socket.connect
original_connect_ex = socket.socket.connect_ex
original_socketpair = socket.socketpair
CONTROL_SOCKETPAIR = ContextVar('trial_asyncio_control_socketpair',default=False)


def control_socketpair(*args, **kwargs):
    token = CONTROL_SOCKETPAIR.set(True)
    try:
        return original_socketpair(*args, **kwargs)
    finally:
        CONTROL_SOCKETPAIR.reset(token)


def guarded_connect(sock, address):
    if CONTROL_SOCKETPAIR.get() and isinstance(address,tuple) and address[0] in ('127.0.0.1','::1'):
        return original_connect(sock,address)
    if not isinstance(address, tuple) or address[0] not in ('127.0.0.1', '::1') or address[1] != 11434:
        raise OSError('Trial transport denied outside the approved local Ollama endpoint')
    return original_connect(sock, address)


def guarded_connect_ex(sock, address):
    if CONTROL_SOCKETPAIR.get() and isinstance(address,tuple) and address[0] in ('127.0.0.1','::1'):
        return original_connect_ex(sock,address)
    if not isinstance(address, tuple) or address[0] not in ('127.0.0.1', '::1') or address[1] != 11434:
        raise OSError('Trial transport denied outside the approved local Ollama endpoint')
    return original_connect_ex(sock, address)

def metadata(path, payload=None):
    with httpx.Client(timeout=min(5, remaining()), trust_env=False) as client:
        with client.stream('POST' if payload else 'GET', ENDPOINT+path, json=payload) as response:
            response.raise_for_status()
            raw = bytearray()
            for chunk in response.iter_bytes():
                raw.extend(chunk)
                if len(raw) > 2 * 1024 * 1024:
                    raise ValueError('Trial metadata byte ceiling exceeded')
    return json.loads(raw)


def model_digest():
    tags = metadata('/api/tags')
    entry = next(row for row in tags['models'] if row.get('name', row.get('model')) == MODEL)
    return entry['digest']


original_stream = OllamaProvider._stream_chat


async def bounded_stream(self, model, payload, timeout):
    left = remaining()
    if model != MODEL or self._base_url != ENDPOINT:
        raise TrialLimitReached('Trial model/endpoint changed')
    row = {
        'attempt': len(ATTEMPTS)+1, **CASE_CONTEXT,
        'started_seconds': round(time.monotonic()-START, 3),
        'model': model, 'num_ctx_requested': payload['options']['num_ctx'],
        'num_predict': payload['options']['num_predict'], 'think_sent': payload.get('think', 'omitted'),
        'input_image_hashes': [hashlib.sha256(base64.b64decode(image)).hexdigest() for image in payload['messages'][0].get('images', [])],
        'status': 'started',
    }
    ATTEMPTS.append(row)
    save()
    try:
        response, trace_present = await original_stream(self, model, payload, min(timeout, left))
        row.update({'status': 'completed', 'input_tokens': response.input_tokens, 'output_tokens': response.output_tokens, 'finish_reason': response.finish_reason, 'trace_present': trace_present})
        return response, trace_present
    except Exception as exc:
        row.update({'status': 'failed', 'error_type': type(exc).__name__})
        raise
    finally:
        row['ended_seconds'] = round(time.monotonic()-START, 3)
        save()


async def fixed_context(self, model, prompt, max_tokens):
    advertised = await self.get_context_window(model)
    if advertised < 16384:
        raise ValueError('Trial model cannot support the predeclared context setting')
    return 16384


def bounded_call(config_arg, workload_tag, prompt, **kwargs):
    left = remaining()
    if config_arg.resolve(workload_tag) != ('ollama', MODEL):
        raise ValueError('Trial route is not the approved local model')
    prompt_hash = hashlib.sha256(prompt.encode()).hexdigest()
    atomic_write_text(ROOT/'prompts'/f'{prompt_hash}.txt', prompt)
    kwargs.update({'max_tokens': min(kwargs.get('max_tokens', 1536), 1536), 'timeout': max(1, min(60, int(left))), 'retries': 0, 'temperature': 0})
    row = {**CASE_CONTEXT, 'workload_tag': workload_tag, 'call_type': kwargs.get('call_type',''), 'prompt_sha256': prompt_hash, 'started_seconds': round(time.monotonic()-START,3), 'status': 'started'}
    CALLS.append(row)
    save()
    try:
        response = canonical_call(config_arg, workload_tag, prompt, **kwargs)
        if response.provider_name != 'ollama' or response.provider_type != 'local':
            raise ValueError('Trial response did not retain proven local route identity')
        row.update({'status': 'completed', 'response': asdict(response)})
        return response
    except Exception as exc:
        row.update({'status': 'failed', 'error_type': type(exc).__name__})
        raise
    finally:
        row['ended_seconds'] = round(time.monotonic()-START,3)
        save()

def run_pairs():
    source_plan = {
        'fixtures': {key: asdict(value) for key,value in fixtures.items()},
        'screening_pool': [asdict(paper) for paper in papers], 'screening_query': screen_query,
        'repeats': 2, 'settings': [False,True], 'cache': 'disabled',
        'context_override': 'Trial-only provider transport override pins requested num_ctx16384 for allconditions; production adaptive context is not validated by this trial.',
        'screening_note': 'Production paper_rerank is structured JSON and sends thinkfalse for BOTH requests. No effective reasoning contrast is expected; never manufacture one.',
    }
    atomic_write_json(ROOT/'source-plan.json', source_plan)
    names = ['faros-extraction','mature-source-screening','conflict-correction','freshness-change']
    for repeat in (1,2):
        for index, name in enumerate(names):
            for think in ((False,True) if (repeat+index)%2 else (True,False)):
                remaining()
                if model_digest() != RECEIPT['runtime']['model_digest']:
                    raise ValueError('Model artifact identity changed during the paired trial')
                set_think(think)
                CASE_CONTEXT.clear(); CASE_CONTEXT.update({'case_id':name,'repeat':repeat,'think_requested':think})
                tracker = CostTracker()
                before = len(CALLS)
                row = {**CASE_CONTEXT, 'status':'started'}
                ROWS.append(row); save()
                try:
                    if name == 'mature-source-screening':
                        ranked = ranking.rerank_papers(screen_query,papers,config,tracker=tracker,top_n=1)
                        row['selected'] = [{'paper_id':item.paper.paper_id,'selected_by':item.selected_by,'rationale':item.rationale} for item in ranked]
                        row['output'] = CALLS[-1].get('response',{}).get('text','') if len(CALLS)>before else ''
                    else:
                        row['output'] = harness._run_analysis(fixtures[name],rc,tracker)
                    row['status'] = 'completed' if row['output'].strip() else 'empty_final'
                except Exception as exc:
                    row.update({'status':'failed','error_type':type(exc).__name__,'output':''})
                row['logical_call_indexes'] = list(range(before,len(CALLS)))
                row['input_tokens'] = tracker.total_input_tokens
                row['output_tokens'] = tracker.total_output_tokens
                row['cost_usd'] = tracker.total_cost
                save()


def run_visual():

    CASE_CONTEXT.clear(); CASE_CONTEXT.update({'case_id':'synthetic-local-video','think_requested':False})
    assets = ROOT/'assets'; assets.mkdir(exist_ok=True)
    powershell = resolve_executable('pwsh')

    if not powershell or not ffmpeg:
        raise ValueError('Existing media tools unavailable; no installation permitted')
    subprocess.run([powershell,'-NoProfile','-NonInteractive','-File',str(Path('.agent/issue61_67_assets.ps1').resolve()),'-OutputRoot',str(assets)],check=True,timeout=min(20,remaining()),env=sanitized_package_env(),capture_output=True)
    concat = "file 'initial.png'\nduration65\nfile 'qualification.png'\nduration5\nfile 'initial.png'\nduration55\nfile 'initial.png'\n".replace('duration','duration ')
    (assets/'slides.txt').write_text(concat,encoding='utf-8')
    source = assets/'synthetic.mp4'
    subprocess.run([ffmpeg,'-v','error','-nostdin','-protocol_whitelist','file,pipe','-f','concat','-safe','0','-i',str(assets/'slides.txt'),'-vf','fps=1','-t','125','-c:v','mpeg4','-q:v','2','-an','-y',str(source)],check=True,timeout=min(30,remaining()),env=sanitized_package_env(),capture_output=True)

    atomic_write_json(ROOT/'grid.json',asdict(grid))
    images = tuple(LLMImage(read_confined_bytes(ROOT/'grid'/frame.relative_path,ROOT/'grid',max_bytes=4*1024*1024)) for frame in grid.frames)
    grid_response = bounded_call(rc,'analysis','Describe only the readable text and visible qualifications in these sampled frames. Name any missing context or unknowns. Do not infer unseen slides.',images=images,call_type='video_grid_probe',usage_tracker=CostTracker())
