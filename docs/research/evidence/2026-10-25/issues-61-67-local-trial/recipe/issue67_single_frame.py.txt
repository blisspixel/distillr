import asyncio
import hashlib
import json
import os
import socket
from contextvars import ContextVar
from dataclasses import asdict
from datetime import UTC, datetime
from pathlib import Path

from distill.library.confined import read_confined_bytes
from distill.library.paths import atomic_write_json
from distill.llm import call
from distill.llm.images import LLMImage
from distill.llm.providers.ollama import OllamaProvider
from distill.llm.router import RouterConfig
from distill.pipeline.costs import CostTracker,TokenUsage

root=Path('.agent/issue61-67-trial').resolve()
prior=json.loads((root/'trial.json').read_text())
authority=json.loads(Path('.agent/issue61-67-authority.json').read_text())
deadline=datetime.fromisoformat(authority['deadline_utc'])
base_count=len(prior['transport_attempts'])
assert base_count==21
attempts=[]
local_pair=ContextVar('control_pair',default=False)
old_pair=socket.socketpair
old_connect=socket.socket.connect
old_connect_ex=socket.socket.connect_ex

def pair(*args,**kwargs):
    token=local_pair.set(True)
    try:return old_pair(*args,**kwargs)
    finally:local_pair.reset(token)

def connect(sock,address):
    if not isinstance(address,tuple) or address[0] not in ('127.0.0.1','::1') or (address[1]!=11434 and not local_pair.get()):
        raise OSError('Diagnostic outbound route denied')
    return old_connect(sock,address)

def connect_ex(sock,address):
    if not isinstance(address,tuple) or address[0] not in ('127.0.0.1','::1') or (address[1]!=11434 and not local_pair.get()):
        raise OSError('Diagnostic outbound route denied')
    return old_connect_ex(sock,address)

socket.socketpair=pair
socket.socket.connect=connect
socket.socket.connect_ex=connect_ex
os.environ.update({'OLLAMA_BASE_URL':'http://127.0.0.1:11434','DISTILL_ANALYSIS_LOCAL_THINK':'false','HF_HUB_OFFLINE':'1','TRANSFORMERS_OFFLINE':'1'})
old_stream=OllamaProvider._stream_chat

async def bounded_stream(self,model,payload,timeout):
    left=(deadline-datetime.now(UTC)).total_seconds()
    if left<1 or base_count+len(attempts)>=24:
        raise ValueError('Approved shared request/time limit reached')
    attempts.append({'attempt':base_count+len(attempts)+1,'think_sent':payload.get('think'),'num_ctx_requested':payload['options']['num_ctx']})
    response,trace=await old_stream(self,model,payload,min(timeout,left))
    attempts[-1].update({'trace_present':trace,'finish_reason':response.finish_reason,'input_tokens':response.input_tokens,'output_tokens':response.output_tokens})
    return response,trace

async def fixed_context(self,model,prompt,max_tokens):
    if await self.get_context_window(model)<16384:raise ValueError('Context identity changed')
    return 16384

OllamaProvider._stream_chat=bounded_stream
OllamaProvider._adaptive_num_ctx=fixed_context
frames=json.loads((root/'targeted.json').read_text())
frame=next(row for row in frames['frames'] if row['timestamp_ms']==67000)
data=read_confined_bytes(root/'targeted'/frame['relative_path'],root/'targeted',max_bytes=4*1024*1024)
assert data is not None and hashlib.sha256(data).hexdigest()==frame['sha256']
image=LLMImage(data)
tracker=CostTracker()
result={'schema':'single-frame-diagnostic.v1','purpose':'Diagnose actual pixel reading after sampled-frame hallucination; no expected text is supplied to model.','base_attempts':base_count,'attempts':attempts,'frame_sha256':image.sha256,'status':'started'}
try:
    response=call(RouterConfig(provider='ollama',model='gemma4:e2b',analysis_provider='ollama',cost_mode='no-metered',fallback_provider='',fallback_model=''),'analysis','Transcribe every readable word from the supplied PNG image. Preserve the written qualifications exactly. If the text cannot be read, say UNKNOWN. Do not infer a table, data values, or text from prior conversation.',images=(image,),max_tokens=512,retries=0,timeout=max(1,min(30,int((deadline-datetime.now(UTC)).total_seconds()))),temperature=0,call_type='video_single_frame_diagnostic',usage_tracker=tracker)
    tracker.record(TokenUsage.from_response(response,call_type='video_single_frame_diagnostic'))
    result.update({'status':'completed','response':asdict(response)})
except Exception as exc:
    result.update({'status':'failed','error_type':type(exc).__name__})
finally:
    result['total_approved_attempts']=base_count+len(attempts)
    result['completed_at']=datetime.now(UTC).isoformat()
    atomic_write_json(root/'single-frame-diagnostic.json',result)
    print(json.dumps({'status':result['status'],'total_attempts':result['total_approved_attempts'],'output':result.get('response',{}).get('text','')}))
