import os
import subprocess
import sys
import json
from datetime import UTC, datetime
from pathlib import Path

from distill.process_resources import ProcessBudgetExceeded, terminate_isolated_process_tree, wait_for_process_budget
from distill.process_security import sanitized_package_env

with Path('.agent/issue61-67-trial-console.log').open('wb') as output:
    process = subprocess.Popen(
        [sys.executable, str(Path(sys.argv[1] if len(sys.argv)>1 else '.agent/issue61_67_trial.py').resolve())],
        stdin=subprocess.DEVNULL, stdout=output, stderr=subprocess.STDOUT,
        env=sanitized_package_env(), start_new_session=os.name != 'nt',
        creationflags=(subprocess.CREATE_NEW_PROCESS_GROUP | subprocess.CREATE_NO_WINDOW) if os.name == 'nt' else 0,
    )
    try:
        authority = json.loads(Path('.agent/issue61-67-authority.json').read_text())
        left = datetime.fromisoformat(authority['deadline_utc']).timestamp() - datetime.now(UTC).timestamp()
        code = wait_for_process_budget(process, timeout_seconds=max(0.01,left), memory_limit_bytes=1024**3)
        print(f'Authorized trial process exit: {code}')
    except ProcessBudgetExceeded as exc:
        terminate_isolated_process_tree(process)
        print(f'Authorized trial stopped by outer {exc.kind} budget')
